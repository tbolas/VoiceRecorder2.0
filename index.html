function submitRecording() { if (!audioBlob) { showToast('No recording found.', true); return; } document.getElementById('submit-btn').disabled = true; document.getElementById('submit-btn').textContent = 'Saving...'; showToast('Converting audio...'); blobToBase64(audioBlob).then(function(base64) { var dataURI = 'data:' + audioBlob.type + ';base64,' + base64; showToast('Sending ' + Math.round(dataURI.length / 1024) + 'KB...'); var payload = { action: 'saveRecording', row: currentItem.row, volume: currentItem.volume, number: currentItem.number, text: currentItem.text, value: currentItem.value, user: userData.user, lumicash: userData.lumicash, dataURI: dataURI }; return fetch(APPS_SCRIPT_URL, { method: 'POST', mode: 'no-cors', body: JSON.stringify(payload) }); }).then(function(response) { showPanel('done-panel'); document.getElementById('done-msg').textContent = 'Saved! (' + currentItem.value + ' BIF earned)'; document.querySelector('#done-panel .btn').style.display = 'inline-block'; }).catch(function(err) { showToast('Error: ' + err.message, true); document.getElementById('submit-btn').disabled = false; document.getElementById('submit-btn').textContent = 'Submit'; }); }
