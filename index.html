const SHEET_ID = '1sX_-P3Co1h1y1miOnAWKv7SS5Rvl2_-s9_On9bUpA6A'; const DRIVE_FOLDER_NAME = 'VoiceRecorder2.0'; function doGet(e) { var callback = e.parameter.callback; var action = e.parameter.action; var result; try { if (action === 'getNextItem') { result = getNextItem(); } else { result = { error: 'Unknown action: ' + action }; } } catch (err) { result = { error: err.toString() }; } var output = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + '(' + output + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var output = ContentService.createTextOutput('ok'); output.setMimeType(ContentService.MimeType.TEXT); try { var data = JSON.parse(e.postData.contents); if (data.action === 'submit') saveRecording(data); } catch(err) {} return output; } function getNextItem() { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheetByName('ToRecord'); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][3]).toLowerCase() === 'available') { sheet.getRange(i + 1, 4).setValue('Pending'); return { row: i + 1, volume: String(data[i][0]), number: String(data[i][1]), text: String(data[i][2]), value: data[i][4] }; } } return { done: true }; } function saveRecording(data) { var ss = SpreadsheetApp.openById(SHEET_ID); var toRecordSheet = ss.getSheetByName('ToRecord'); var now = new Date(); var ts = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyyMMdd_HHmmss'); var tsDisplay = Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss'); var safeName = data.volume + '_' + data.number + '_' + data.user + '_' + ts; var folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME); var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME); // Save audio - identical to 1.0 var byteString = Utilities.base64Decode(data.dataURI.split(',')[1]); var mimeString = data.dataURI.split(',')[0].split(':')[1].split(';')[0]; var blob = Utilities.newBlob(byteString, mimeString); blob.setName(safeName + '.mp3'); var audioFile = folder.createFile(blob); var mp3Link = audioFile.getUrl(); // Save TXT folder.createFile(Utilities.newBlob(data.text, 'text/plain', safeName + '.txt')); // Mark Done toRecordSheet.getRange(Number(data.row), 4).setValue('Done'); // Append to Audit var auditSheet = ss.getSheetByName('Audit'); if (!auditSheet) { auditSheet = ss.insertSheet('Audit'); auditSheet.appendRow(['Volume', 'Number', 'Text', 'Value', 'User', 'Lumicash', 'TimeComplete', 'MP3Link']); } auditSheet.appendRow([ data.volume, data.number, data.text, data.value, data.user, data.lumicash, tsDisplay, mp3Link ]); // Update Payment updatePayment(ss, data.user, data.lumicash); } function updatePayment(ss, user, lumicash) { var paySheet = ss.getSheetByName('Payment'); if (!paySheet) { paySheet = ss.insertSheet('Payment'); paySheet.appendRow(['User', 'Lumicash', 'Total Earned', 'Total Paid', 'Total Owed']); } var auditData = ss.getSheetByName('Audit').getDataRange().getValues(); var totalEarned = 0; for (var i = 1; i < auditData.length; i++) { if (String(auditData[i][4]) === String(user)) { totalEarned += Number(auditData[i][3]) || 0; } } var payData = paySheet.getDataRange().getValues(); var userRow = -1; for (var i = 1; i < payData.length; i++) { if (String(payData[i][0]) === String(user)) { userRow = i + 1; break; } } if (userRow === -1) { paySheet.appendRow([user, lumicash, totalEarned, 0, totalEarned]); } else { var totalPaid = Number(payData[userRow - 1][3]) || 0; paySheet.getRange(userRow, 2).setValue(lumicash); paySheet.getRange(userRow, 3).setValue(totalEarned); paySheet.getRange(userRow, 5).setValue(totalEarned - totalPaid); } }
